% Labels sit on their baseline, placed half a cap height below the given y, so every word in a
% row shares one baseline. (Centring the box instead moves words with accents or descenders.)
% The boxes are built outside the picture: inside it TikZ typesets loose text in \nullfont.
\newcommand\SheetSetCapShift[1]{\pgfinterruptpicture\global\setbox\SheetCapBox\hbox{#1H}%
  \global\SheetCapShift=.5\ht\SheetCapBox\endpgfinterruptpicture}
\newcommand\SheetBaseNode[4]{\SheetSetCapShift{#3}%
  \node[anchor=base west,inner sep=0pt,text=sheetink,font=#3,yshift=-\SheetCapShift] at (#1,#2) {#4};}
\newcommand\SheetBold[4]{\SheetBaseNode{#1}{#2}{\RPGBodyBold\fontsize{#3}{#3}\selectfont}{#4}}
\newcommand\SheetSmall[4]{\SheetBaseNode{#1}{#2}{\RPGBodyFont\fontsize{#3}{#3}\selectfont}{#4}}
% Numerals in the hit-point boxes stay optically centred on their box.
\newcommand\SheetCenter[4]{\node[anchor=center,inner sep=0pt,text=sheetink,font=\RPGBodyBold\fontsize{#3}{#3}\selectfont] at (#1,#2) {#4};}
% \SheetFit{base anchor}{x}{y}{font}{free width in mm}{text}: shrinks the text proportionally
% only when it is wider than the free space measured on the background artwork.
\newcommand\SheetFit[6]{\pgfinterruptpicture\global\setbox\SheetMeasure\hbox{\color{sheetink}#4#6}%
  \global\setbox\SheetCapBox\hbox{#4H}\global\SheetCapShift=.5\ht\SheetCapBox
  \ifdim\wd\SheetMeasure>#5mm\relax
    \pgfmathsetmacro\SheetFitRatio{#5*72.27/25.4/\wd\SheetMeasure}%
    \global\SheetCapShift=\SheetFitRatio\SheetCapShift
    \global\setbox\SheetMeasure\hbox{\resizebox{#5mm}{!}{\box\SheetMeasure}}\fi
  \endpgfinterruptpicture\node[anchor=#1,inner sep=0pt,yshift=-\SheetCapShift] at (#2,#3) {\usebox\SheetMeasure};}
% Both section headings share one size per language: at most 12 pt, shrunk together only when
% either heading would be wider than its free span between the ornaments.
\newcommand\SheetComputeHeadingSize{%
  \sbox\SheetMeasure{\RPGBodyBold\fontsize{12}{12}\selectfont\Sh{Attributes}}%
  \pgfmathsetmacro\SheetHeadA{33.5/25.4*72.27/\wd\SheetMeasure*12}%
  \sbox\SheetMeasure{\RPGBodyBold\fontsize{12}{12}\selectfont\Sh{Abilities}}%
  \pgfmathsetmacro\SheetHeadB{29.2/25.4*72.27/\wd\SheetMeasure*12}%
  \pgfmathsetmacro\SheetHeadC{min(12,\SheetHeadA,\SheetHeadB)}\xdef\SheetHeadingSize{\SheetHeadC}}
\newcommand\SheetHeadingFont{\RPGBodyBold\fontsize{\SheetHeadingSize}{\SheetHeadingSize}\selectfont}
% Tier labels: one common size per language (at most 7 pt), so the four columns match.
% Label x = checkbox right edge + 1.5 mm; free width ends 1.5 mm before the next box or the frame.
\newcommand\SheetTierFont{\RPGBodyFont\fontsize{7}{7}\selectfont}
\def\SheetTierX#1{\ifcase#1\or116.06\or138.58\or162.09\or181.81\fi}
\def\SheetTierFree#1{\ifcase#1\or15.73\or16.92\or13.14\or15.93\fi}
\newcommand\SheetComputeTierScale{\gdef\SheetTierScale{1}%
  \foreach \k in {1,...,4}{\sbox\SheetMeasure{\SheetTierFont\Sh{Tier\k}}%
    \pgfmathsetmacro\fit{min(\SheetTierScale,\SheetTierFree{\k}*72.27/25.4/\wd\SheetMeasure)}%
    \xdef\SheetTierScale{\fit}}}
% Tier labels share the baseline of the attribute name in the same row (8.4 pt label font).
\newcommand\SheetTiers[1]{\SheetSetCapShift{\RPGBodyBold\fontsize{8.4}{8.4}\selectfont}%
  \foreach \k in {1,...,4}{\node[anchor=base west,inner sep=0pt,text=sheetink,yshift=-\SheetCapShift] at (\SheetTierX{\k},#1)
  {\scalebox{\SheetTierScale}{\SheetTierFont\Sh{Tier\k}}};}}
